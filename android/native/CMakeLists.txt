cmake_minimum_required(VERSION 3.22.1)
# Gradle's Debug variant would otherwise build whisper/ggml/bridge without -O. Keep symbols, optimise.
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CMAKE_BUILD_TYPE "RelWithDebInfo" CACHE STRING "Native build type" FORCE)
endif()
project(parla_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# arm64: ggml's runtime-selected CPU variants (bridge.cpp loads the best this CPU can run), so fp16/dotprod
# phones are not held to armv8.0 kernels. That needs shared ggml libraries. x86_64 (emulator) stays static.
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(PARLA_CPU_VARIANTS ON)
else()
    set(PARLA_CPU_VARIANTS OFF)
endif()
set(BUILD_SHARED_LIBS ${PARLA_CPU_VARIANTS} CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ${PARLA_CPU_VARIANTS} CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ${PARLA_CPU_VARIANTS} CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
foreach(flag WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_CURL WHISPER_COREML WHISPER_OPENVINO GGML_NATIVE GGML_OPENMP GGML_CUDA GGML_METAL GGML_VULKAN GGML_OPENCL GGML_SYCL GGML_BLAS GGML_ACCELERATE GGML_SSE42 GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_AVX512 GGML_AVX_VNNI GGML_LLAMAFILE)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(vendor/whisper)
add_library(parla-whisper SHARED bridge.cpp)
target_link_libraries(parla-whisper PRIVATE whisper android log)
if(PARLA_CPU_VARIANTS)
    target_compile_definitions(parla-whisper PRIVATE PARLA_CPU_VARIANTS)
endif()
target_link_options(parla-whisper PRIVATE "-Wl,-z,max-page-size=16384")
