cmake_minimum_required(VERSION 3.22.1)
project(parla_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_CURL WHISPER_COREML WHISPER_OPENVINO GGML_NATIVE GGML_OPENMP GGML_CUDA GGML_METAL GGML_VULKAN GGML_OPENCL GGML_SYCL GGML_BLAS GGML_ACCELERATE GGML_BACKEND_DL GGML_CPU_ALL_VARIANTS GGML_SSE42 GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_AVX512 GGML_AVX_VNNI GGML_LLAMAFILE)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
add_subdirectory(vendor/whisper)
add_library(parla-whisper SHARED bridge.cpp)
target_link_libraries(parla-whisper PRIVATE whisper android log)
target_link_options(parla-whisper PRIVATE "-Wl,-z,max-page-size=16384")
